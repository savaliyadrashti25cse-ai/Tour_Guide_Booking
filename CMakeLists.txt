cmake_minimum_required(VERSION 3.16)
project(TourGuideBooking LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS "Source/*.cpp")
list(REMOVE_ITEM CORE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/Source/main.cpp")
add_library(tourguide_core ${CORE_SOURCES})
target_include_directories(tourguide_core PUBLIC Source)
add_executable(Project Source/main.cpp)
target_link_libraries(Project PRIVATE tourguide_core)
add_executable(system_tests Tests/system_tests.cpp)
target_link_libraries(system_tests PRIVATE tourguide_core)
enable_testing()
add_test(NAME booking_system COMMAND system_tests)
